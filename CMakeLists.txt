cmake_minimum_required(VERSION 3.25)
project(open_dynamic_rpg LANGUAGES CXX)

option(ODR_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(ODR_SHARED "Build the Unreal-facing core as a shared library" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(CTest)
set(JSON_BuildTests OFF CACHE INTERNAL "")
set(JSON_Install OFF CACHE INTERNAL "")
add_subdirectory(third_party/nlohmann_json)

if(ODR_SHARED)
    add_library(odr_core SHARED core/src/session.cpp)
else()
    add_library(odr_core STATIC core/src/session.cpp)
endif()
target_include_directories(odr_core PUBLIC core/include)
target_link_libraries(odr_core PRIVATE nlohmann_json::nlohmann_json)
set_target_properties(odr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

function(odr_owned_target target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
        if(ODR_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

odr_owned_target(odr_core)

add_executable(odr_tool tools/odr_tool.cpp)
target_link_libraries(odr_tool PRIVATE odr_core nlohmann_json::nlohmann_json)
odr_owned_target(odr_tool)

if(BUILD_TESTING)
    add_executable(odr_tests tests/session_tests.cpp)
    target_link_libraries(odr_tests PRIVATE odr_core nlohmann_json::nlohmann_json)
    odr_owned_target(odr_tests)
    add_test(NAME session COMMAND odr_tests)
    set_tests_properties(session PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
